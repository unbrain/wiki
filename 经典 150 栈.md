
[20. 有效的括号](https://leetcode.cn/problems/valid-parentheses/)

```javascript
var isValid = function(s) {
    const map = new Map()
    map.set('(',')')
    map.set('[',']')
    map.set('{','}')
    const res = []
    for(let char of s) {
        if(map.has(char)) {
            res.push(char)
        } else if(char === map.get(res[res.length - 1])) {
            res.pop()
        } else {
            return false
        }
    }
    return res.length === 0
};
```
