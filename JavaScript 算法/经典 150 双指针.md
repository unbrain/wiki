[125. 验证回文串](https://leetcode.cn/problems/valid-palindrome/)

```javascript
var isPalindrome = function(s) {
    s = s.trim()
    if(!s.length) return true
    
    // 关键修改：增加 0-9 匹配数字
    let arr = s.toLowerCase().match(/[a-z0-9]/g) 
    
    if(!arr) return true
    let left = 0
    let right = arr.length-1
    while(left < right) {
        if(arr[left] === arr[right]) {
            left++
            right--
        } else {
            return false
        }
    }
    return true
};
```
