[228. 汇总区间](https://leetcode.cn/problems/summary-ranges/)

> 将有序无重复整数数组按连续区间分组，每个区间表示为 `"a"` 或 `"a->b"`。

```javascript
var summaryRanges = function(nums) {
  const res = []
  let i = 0
  while (i < nums.length) {
    // 当前区间的起点
    const start = nums[i]
    // 向右扩展：只要下一个数比当前大 1，就移动 i
    while (i + 1 < nums.length && nums[i + 1] === nums[i] + 1)
      i++
    // i 停留在区间终点，根据长度决定输出格式
    if (start === nums[i])
      res.push(`${start}`)
    else
      res.push(`${start}->${nums[i]}`)
    i++
  }
  return res
}
```
