[141. 环形链表](https://leetcode.cn/problems/linked-list-cycle/)

```javascript
var hasCycle = function(head) {
    let slow = head
    let fast = head
    while(fast?.next) {
        slow = slow.next
        fast = fast.next.next
        if(slow === fast) {
            return true
        }
    }

    return false
};
```

[2. 两数相加](https://leetcode.cn/problems/add-two-numbers/)

```javascript
var addTwoNumbers = function(l1, l2) {
    let dummy = new ListNode()
    let p1 = l1, p2 = l2, p3 = dummy
    let count = 0

    while(p1 || p2) {
        let temp = (p1?.val??0) + (p2?.val??0) + count
        p3.next = new ListNode(temp % 10)
        count = temp > 9 ? 1 : 0
        p1 = p1?.next
        p2 = p2?.next
        p3 = p3?.next
    }
    if(count) {
        p3.next = new ListNode(count)
    }

    return dummy.next
};
```

[21. 合并两个有序链表](https://leetcode.cn/problems/merge-two-sorted-lists/)

```javascript
var mergeTwoLists = function(list1, list2) {
    let dummy = new ListNode()
    let p1 = list1, p2 = list2, p3 = dummy

    while(p1 && p2) {
        if(p1.val < p2.val) {
            p3.next = p1
            p1 = p1.next
        } else {
            p3.next = p2
            p2 = p2.next
        }
        p3 = p3.next
    }

    p3.next = p1 ? p1 : p2

    return dummy.next
};
```

[92. 反转链表 II](https://leetcode.cn/problems/reverse-linked-list-ii/)

```javascript
var reverseBetween = function(head, left, right) {
    if(left === right) return head
    let dummy = new ListNode(null, head)
    let past = dummy

    for(let i = 1; i < left; i++) {
        past = past.next
    }
    let current = past.next
    let prev = null

    for(let i = left; i<= right;i++) {
        let temp = current.next
        current.next = prev
        prev = current
        current = temp
    }

    past.next.next = current
    past.next = prev

    return dummy.next
};
```
