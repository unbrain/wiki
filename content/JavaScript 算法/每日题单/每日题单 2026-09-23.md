```javascript
/**
 * 快速排序（Hoare 双指针对撞分区 + 随机基准防退化）
 * @param {number[]} nums
 * @return {number[]}
 */
var sortArray = function(nums) {
    function quickSort(l, r) {
        if (l >= r) return;
        const pivot = nums[Math.floor(Math.random() * (r - l + 1)) + l];
        let i = l, j = r;
        while (i <= j) {
            while (nums[i] < pivot) i++;
            while (nums[j] > pivot) j--;
            if (i <= j) {
                [nums[i], nums[j]] = [nums[j], nums[i]];
                i++;
                j--;
            }
        }
        quickSort(l, j);
        quickSort(i, r);
    }
    quickSort(0, nums.length - 1);
    return nums;
};
```

```javascript
/**
 * 快速选择 QuickSelect（Hoare 双指针分区 + 随机基准 + 门神定位法）
 * @param {number[]} nums
 * @param {number} k
 * @return {number}
 */
var findKthLargest = function(nums, k) {
    const target = nums.length - k;

    const qs = (l, r) => {
        if (l >= r) return nums[target];

        const randomIndex = Math.floor(Math.random() * (r - l + 1)) + l;
        const pivot = nums[randomIndex];

        let i = l, j = r;
        while (i <= j) {
            while (nums[i] < pivot) i++;
            while (nums[j] > pivot) j--;
            if (i <= j) {
                [nums[i], nums[j]] = [nums[j], nums[i]];
                i++;
                j--;
            }
        }

        // 门神法则：入左院、进右院、夹中间直取果
        if (target <= j) return qs(l, j);
        if (target >= i) return qs(i, r);
        return nums[target];
    };

    return qs(0, nums.length - 1);
};
```
