- [x] 新题 1：[112. 路径总和](https://leetcode.cn/problems/path-sum/)

```javascript
/**
 * 递归（差值下传法）/ 叶子节点判定与短路剪枝
 * @param {TreeNode} root
 * @param {number} targetSum
 * @return {boolean}
 */
var hasPathSum = function(root, targetSum) {
    if(root === null) return false
    if(root.left === null && root.right === null) {
        return targetSum === root.val
    }
    const reminding = targetSum - root.val
    return hasPathSum(root.left, reminding) || hasPathSum(root.right, reminding)
};
```
