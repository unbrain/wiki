```javascript
/**
 * 双指针中心扩散法（实时结算 + 索引优化版）
 * @param {string} s
 * @return {string}
 */
var longestPalindrome = function(s) {
    if (s.length < 2) return s;

    let start = 0, maxLen = 1;

    const expand = (i, j) => {
        while (i >= 0 && j < s.length && s[i] === s[j]) {
            const curLen = j - i + 1;
            if (curLen > maxLen) {
                maxLen = curLen;
                start = i;
            }
            i--;
            j++;
        }
    };

    for (let i = 0; i < s.length; i++) {
        expand(i, i);     // 奇数中心
        expand(i, i + 1); // 偶数中心
    }

    return s.slice(start, start + maxLen);
};
```
